Program Example for all printers
Print Sample
PRINT #1, CHR$(&H1B);"t";CHR$(0); ← Select page 0
GOSUB printing
PRINT #1, CHR$(&H1B);"t";CHR$(1); ← Select page 1
GOSUB printing
END
printing:
  FOR i=&H20 TO &H7F
    PRINT #1, CHR$(i);
  NEXT i
  PRINT #1, CHR$(&HA);
  FOR i=&H80 TO &HFF
    PRINT #1, CHR$(i);
  NEXT i
  PRINT #1, CHR$(&HA);
  RETURN
